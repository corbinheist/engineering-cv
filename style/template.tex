\usepackage{fontspec}

% Fallback chain: Helvetica Neue → TeX Gyre Heros → Arial
\IfFontExistsTF{Helvetica Neue}{%
  \setmainfont{Helvetica Neue}[
    BoldFont       = Helvetica Neue Bold,
    ItalicFont     = Helvetica Neue Italic,
    BoldItalicFont = Helvetica Neue Bold Italic
  ]
  \newfontfamily\lightfont{Helvetica Neue Light}
}{%
  \IfFontExistsTF{TeX Gyre Heros}{%
    \setmainfont{TeX Gyre Heros}[
      BoldFont       = TeX Gyre Heros Bold,
      ItalicFont     = TeX Gyre Heros Italic,
      BoldItalicFont = TeX Gyre Heros Bold Italic
    ]
    \newfontfamily\lightfont{TeX Gyre Heros}
  }{%
    \setmainfont{Arial}[
      BoldFont       = Arial Bold,
      ItalicFont     = Arial Italic,
      BoldItalicFont = Arial Bold Italic
    ]
    \newfontfamily\lightfont{Arial}
  }%
}
